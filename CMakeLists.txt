cmake_minimum_required(VERSION 3.22)
# One CLion project, one explicitly selected firmware per build directory.
set(ACTIVE_BOARD master CACHE STRING "master, slave1..slave5, slave6 lamp test, slave7 key test, or motor")
set_property(CACHE ACTIVE_BOARD PROPERTY STRINGS master slave1 slave2 slave3 slave4 slave5 slave6 slave7 motor)
if(NOT ACTIVE_BOARD MATCHES "^(master|slave[1-7]|motor)$")
    message(FATAL_ERROR "Invalid ACTIVE_BOARD: ${ACTIVE_BOARD}")
endif()
if(DEFINED ENERGY_CONFIGURED_BOARD AND NOT ENERGY_CONFIGURED_BOARD STREQUAL ACTIVE_BOARD)
    message(FATAL_ERROR "Use a separate build directory/preset for each MCU. This directory belongs to ${ENERGY_CONFIGURED_BOARD}.")
endif()
set(ENERGY_CONFIGURED_BOARD "${ACTIVE_BOARD}" CACHE INTERNAL "Build directory MCU identity")
set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_SOURCE_DIR}/cmake/arm-none-eabi.cmake" CACHE FILEPATH "ARM toolchain")
project(EnergyControl LANGUAGES C ASM)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(ENERGY_EXECUTABLE_NAME "EnergyFirmware_${ACTIVE_BOARD}_${CMAKE_BUILD_TYPE}")
add_subdirectory("wanwanwanfufu/dbus_switch_energy/fu_${ACTIVE_BOARD}" board)
# A target unique to each board/configuration makes CLion select only compatible
# profiles. CONFIG_NAME alone does not bind the current CLion profile switcher.
set_target_properties(${ENERGY_EXECUTABLE_NAME} PROPERTIES OUTPUT_NAME EnergyFirmware)
add_custom_target(EnergyFirmware DEPENDS ${ENERGY_EXECUTABLE_NAME})
if(ACTIVE_BOARD STREQUAL "master")
    set(ENERGY_DEVICE STM32F407IG)
    set(ENERGY_OPENOCD_TARGET target/stm32f4x.cfg)
else()
    set(ENERGY_DEVICE STM32F103C8)
    set(ENERGY_OPENOCD_TARGET target/stm32f1x.cfg)
endif()
configure_file(cmake/firmware.json.in firmware.json @ONLY)
message(STATUS "EnergyControl: ${ACTIVE_BOARD} -> ${ENERGY_DEVICE}, ${CMAKE_BINARY_DIR}/EnergyFirmware.elf")
